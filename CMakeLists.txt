cmake_minimum_required(VERSION 3.20)
project(AddonDiscord VERSION 1.0.0 LANGUAGES CXX)

option(DISCORD_BUILD_TESTS "Build offline unit tests and the optional connectivity probe" ON)

if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/SDK/CMakeLists.txt")
    message(FATAL_ERROR "Initialize Addon-SDK: git submodule update --init --recursive")
endif()
add_subdirectory(SDK)
include(SDK/cmake/Plugin.cmake)
find_package(Threads REQUIRED)

add_library(DiscordJson STATIC src/Json.cpp)
target_compile_features(DiscordJson PUBLIC cxx_std_17)
target_include_directories(DiscordJson PUBLIC src)
set_target_properties(DiscordJson PROPERTIES POSITION_INDEPENDENT_CODE ON CXX_EXTENSIONS OFF)

if(WIN32)
    set(DISCORD_TRANSPORT src/TransportWin.cpp)
else()
    find_package(Boost 1.74 REQUIRED)
    find_package(OpenSSL REQUIRED)
    set(DISCORD_TRANSPORT src/TransportLinux.cpp)
endif()

add_library(DiscordCore STATIC src/Client.cpp ${DISCORD_TRANSPORT})
target_compile_features(DiscordCore PUBLIC cxx_std_17)
target_include_directories(DiscordCore PUBLIC src)
target_link_libraries(DiscordCore PUBLIC DiscordJson Threads::Threads)
set_target_properties(DiscordCore PROPERTIES POSITION_INDEPENDENT_CODE ON CXX_EXTENSIONS OFF)

if(MSVC)
    target_compile_options(DiscordJson PRIVATE /W4 /utf-8 /permissive-)
    target_compile_options(DiscordCore PRIVATE /W4 /utf-8 /permissive-)
    target_compile_definitions(DiscordCore PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
else()
    target_compile_options(DiscordJson PRIVATE -Wall -Wextra)
    target_compile_options(DiscordCore PRIVATE -Wall -Wextra)
endif()

if(WIN32)
    target_link_libraries(DiscordCore PUBLIC winhttp)
else()
    target_include_directories(DiscordCore SYSTEM PRIVATE ${Boost_INCLUDE_DIRS})
    target_compile_definitions(DiscordCore PRIVATE BOOST_ERROR_CODE_HEADER_ONLY BOOST_SYSTEM_NO_LIB)
    target_link_libraries(DiscordCore PUBLIC OpenSSL::SSL OpenSSL::Crypto)
endif()

legacy_plugin(Addon-Discord src/Main.cpp)
target_link_libraries(Addon-Discord PRIVATE DiscordCore)

if(DISCORD_BUILD_TESTS)
    enable_testing()
    add_executable(LoadTests tests/LoadTests.cpp)
    target_link_libraries(LoadTests PRIVATE Legacy::SDK ${CMAKE_DL_LIBS})
    add_dependencies(LoadTests Addon-Discord)
    add_test(NAME PluginExports COMMAND LoadTests $<TARGET_FILE:Addon-Discord>)
    add_executable(JsonTests tests/JsonTests.cpp)
    target_link_libraries(JsonTests PRIVATE DiscordJson)
    add_test(NAME JsonTests COMMAND JsonTests)
    add_executable(ClientTests tests/ClientTests.cpp)
    target_link_libraries(ClientTests PRIVATE DiscordCore)
    add_test(NAME ClientTests COMMAND ClientTests)
    add_executable(TransportTests tests/TransportTests.cpp)
    target_link_libraries(TransportTests PRIVATE DiscordCore)
    add_test(NAME TransportLifecycle COMMAND TransportTests)
    set_tests_properties(TransportLifecycle PROPERTIES TIMEOUT 15)
    if(NOT WIN32)
        find_package(Python3 REQUIRED COMPONENTS Interpreter)
        find_program(OPENSSL_EXECUTABLE openssl REQUIRED)
        add_executable(TLSProbe tests/TLSProbe.cpp)
        target_link_libraries(TLSProbe PRIVATE DiscordCore)
        add_test(NAME TLSVerification COMMAND ${Python3_EXECUTABLE}
            ${CMAKE_CURRENT_SOURCE_DIR}/tests/tls_fixture.py $<TARGET_FILE:TLSProbe> ${OPENSSL_EXECUTABLE})
        set_tests_properties(TLSVerification PROPERTIES TIMEOUT 45)
    endif()
    add_executable(LiveProbe tests/LiveProbe.cpp)
    target_link_libraries(LiveProbe PRIVATE DiscordCore)
endif()
